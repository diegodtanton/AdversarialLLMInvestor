def env_bool(
    name: str,
    default: bool,
) -> bool:

    raw = os.getenv(
        name,
        str(default),
    )

    return (
        raw.strip().casefold()
        in {
            "1",
            "true",
            "yes",
            "y",
        }
    )

SCHEMA_SQL = """
PRAGMA journal_mode=WAL;

CREATE TABLE IF NOT EXISTS executions (
    decision_key TEXT PRIMARY KEY,

    trade_date TEXT NOT NULL,
    story_id TEXT NOT NULL,
    ticker TEXT NOT NULL,
    direction TEXT NOT NULL,

    status TEXT NOT NULL,

    requested_notional REAL,

    entry_client_order_id TEXT,
    entry_order_id TEXT,
    requested_qty INTEGER,
    entry_limit_price REAL,

    filled_qty REAL,
    avg_entry_price REAL,
    filled_at TEXT,

    take_profit_price REAL,
    take_profit_client_order_id TEXT,
    take_profit_order_id TEXT,

    exit_deadline TEXT,

    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_executions_status
ON executions(status);

CREATE INDEX IF NOT EXISTS idx_executions_ticker
ON executions(ticker);
"""


def connect_db() -> sqlite3.Connection:
    conn = sqlite3.connect(
        DB_PATH
    )

    conn.row_factory = sqlite3.Row

    conn.executescript(
        SCHEMA_SQL
    )

    return conn

def utc_now_dt() -> datetime:
    return datetime.now(
        timezone.utc
    )


def utc_now() -> str:
    return utc_now_dt().isoformat()


def load_brody() -> dict:
    with INPUT_PATH.open(
        "r",
        encoding="utf-8",
    ) as f:
        return json.load(f)


def decision_key(
    payload: dict,
) -> str:

    trade = payload[
        "selected_trade"
    ]

    return (
        f"{payload['date']}:"
        f"{trade['story_id']}:"
        f"{trade['ticker']}:"
        f"{trade['direction']}"
    )


def client_id_prefix(
    payload: dict,
) -> str:

    trade = payload[
        "selected_trade"
    ]

    date_part = payload[
        "date"
    ].replace(
        "-",
        "",
    )

    story_part = (
        trade[
            "story_id"
        ][-12:]
    )

    ticker = (
        trade[
            "ticker"
        ]
        .lower()
        .replace(
            ".",
            "-",
        )
    )

    return (
        f"brody-{date_part}-"
        f"{story_part}-{ticker}"
    )


def enum_value(value) -> str:
    if value is None:
        return ""

    return str(
        getattr(
            value,
            "value",
            value,
        )
    ).casefold()


def price_increment(
    asset,
) -> Decimal:

    raw = getattr(
        asset,
        "price_increment",
        None,
    )

    if raw:
        return Decimal(
            str(raw)
        )

    # Conventional fallback for listed US stocks.
    return Decimal(
        "0.01"
    )


def round_to_increment(
    price: float,
    increment: Decimal,
    *,
    up: bool,
) -> float:

    p = Decimal(
        str(price)
    )

    units = p / increment

    rounding = (
        ROUND_CEILING
        if up
        else ROUND_FLOOR
    )

    units = units.to_integral_value(
        rounding=rounding
    )

    result = (
        units
        * increment
    )

    return float(result)


def is_terminal_status(
    status,
) -> bool:

    return status in {
        OrderStatus.FILLED,
        OrderStatus.CANCELED,
        OrderStatus.EXPIRED,
        OrderStatus.REJECTED,
        OrderStatus.DONE_FOR_DAY,
        OrderStatus.STOPPED,
        OrderStatus.SUSPENDED,
    }


def is_order_open(
    status,
) -> bool:

    return status in {
        OrderStatus.NEW,
        OrderStatus.ACCEPTED,
        OrderStatus.PENDING_NEW,
        OrderStatus.PARTIALLY_FILLED,
        OrderStatus.ACCEPTED_FOR_BIDDING,
        OrderStatus.PENDING_REPLACE,
        OrderStatus.PENDING_REVIEW,
    }

# QUOTES
# ============================================================

def latest_quote(
    ticker: str,
) -> dict:

    request = StockLatestQuoteRequest(
        symbol_or_symbols=ticker
    )

    result = (
        market_data_client
        .get_stock_latest_quote(
            request
        )
    )

    quote = result[
        ticker
    ]

    bid = float(
        quote.bid_price
        or 0
    )

    ask = float(
        quote.ask_price
        or 0
    )

    if (
        bid <= 0
        or ask <= 0
        or ask < bid
    ):
        raise RuntimeError(
            f"Invalid quote for {ticker}: "
            f"bid={bid}, ask={ask}"
        )

    mid = (
        bid + ask
    ) / 2

    spread_bps = (
        (ask - bid)
        / mid
        * 10000
    )

    return {
        "bid": bid,
        "ask": ask,
        "mid": mid,
        "spread_bps": (
            spread_bps
        ),
    }


def entry_limit_from_quote(
    quote: dict,
    direction: str,
    increment: Decimal,
) -> float:

    cushion = (
        ENTRY_CUSHION_BPS
        / 10000
    )

    if direction == "FOR":
        # Buy limit slightly ABOVE ask.
        #
        # This is marketable but still provides a maximum price.
        raw = (
            quote["ask"]
            * (1 + cushion)
        )

        return round_to_increment(
            raw,
            increment,
            up=True,
        )

    # Short sale.
    #
    # Sell limit slightly BELOW bid.
    # It can execute at the bid or anything better.
    raw = (
        quote["bid"]
        * (1 - cushion)
    )

    return round_to_increment(
        raw,
        increment,
        up=False,
    )

def get_borrow_status(
    asset,
) -> str:

    # New Alpaca field.
    current = getattr(
        asset,
        "borrow_status",
        None,
    )

    if current is not None:
        return enum_value(
            current
        )

    # Compatibility with current alpaca-py versions while
    # easy_to_borrow is being phased out.
    legacy = getattr(
        asset,
        "easy_to_borrow",
        None,
    )

    if legacy is True:
        return "easy_to_borrow"

    if legacy is False:
        return "not_easy_to_borrow"

    return "unknown"


def check_existing_position(
    ticker: str,
) -> None:

    positions = (
        trading_client
        .get_all_positions()
    )

    for position in positions:
        if (
            position.symbol.upper()
            == ticker.upper()
        ):
            raise RuntimeError(
                f"Existing Alpaca position already "
                f"open in {ticker}."
            )


def preflight(
    ticker: str,
    direction: str,
):

    account = (
        trading_client
        .get_account()
    )

    if account.trading_blocked:
        raise RuntimeError(
            "Alpaca account is blocked from trading."
        )

    if (
        float(
            account.buying_power
            or 0
        )
        < TRADE_NOTIONAL_USD
    ):
        raise RuntimeError(
            "Insufficient Alpaca buying power."
        )

    asset = (
        trading_client
        .get_asset(
            ticker
        )
    )

    if not asset.tradable:
        raise RuntimeError(
            f"{ticker} is not tradable on Alpaca."
        )

    if direction == "AGAINST":

        if not asset.shortable:
            raise RuntimeError(
                f"{ticker} is not shortable."
            )

        shorting_enabled = getattr(
            account,
            "shorting_enabled",
            None,
        )

        if shorting_enabled is False:
            raise RuntimeError(
                "Shorting is disabled on this account."
            )

        borrow_status = (
            get_borrow_status(
                asset
            )
        )

        logger.info(
            "%s borrow status: %s",
            ticker,
            borrow_status,
        )

        # First implementation deliberately avoids HTB locates.
        if (
            borrow_status
            != "easy_to_borrow"
        ):
            raise RuntimeError(
                f"{ticker} is not confirmed "
                f"easy-to-borrow: "
                f"{borrow_status}"
            )

    check_existing_position(
        ticker
    )

    clock = (
        trading_client
        .get_clock()
    )

    if not clock.is_open:
        raise RuntimeError(
            "US equity market is currently closed. "
            "Executor will not queue a stale entry overnight."
        )

    return (
        account,
        asset,
    )

def upsert_execution(
    conn: sqlite3.Connection,
    key: str,
    payload: dict,
    *,
    status: str,
    entry_client_order_id: str | None = None,
    entry_order_id: str | None = None,
    requested_qty: int | None = None,
    entry_limit_price: float | None = None,
    filled_qty: float | None = None,
    avg_entry_price: float | None = None,
    filled_at: str | None = None,
    take_profit_price: float | None = None,
    take_profit_client_order_id: str | None = None,
    take_profit_order_id: str | None = None,
    exit_deadline: str | None = None,
) -> None:

    trade = payload[
        "selected_trade"
    ]

    existing = conn.execute(
        """
        SELECT decision_key
        FROM executions
        WHERE decision_key = ?
        """,
        (key,),
    ).fetchone()

    if existing:
        conn.execute(
            """
            UPDATE executions
            SET
                status = ?,
                entry_client_order_id = COALESCE(?, entry_client_order_id),
                entry_order_id = COALESCE(?, entry_order_id),
                requested_qty = COALESCE(?, requested_qty),
                entry_limit_price = COALESCE(?, entry_limit_price),
                filled_qty = COALESCE(?, filled_qty),
                avg_entry_price = COALESCE(?, avg_entry_price),
                filled_at = COALESCE(?, filled_at),
                take_profit_price = COALESCE(?, take_profit_price),
                take_profit_client_order_id = COALESCE(?, take_profit_client_order_id),
                take_profit_order_id = COALESCE(?, take_profit_order_id),
                exit_deadline = COALESCE(?, exit_deadline),
                updated_at = ?
            WHERE decision_key = ?
            """,
            (
                status,
                entry_client_order_id,
                entry_order_id,
                requested_qty,
                entry_limit_price,
                filled_qty,
                avg_entry_price,
                filled_at,
                take_profit_price,
                take_profit_client_order_id,
                take_profit_order_id,
                exit_deadline,
                utc_now(),
                key,
            ),
        )

    else:
        conn.execute(
            """
            INSERT INTO executions (
                decision_key,
                trade_date,
                story_id,
                ticker,
                direction,
                status,
                requested_notional,
                entry_client_order_id,
                entry_order_id,
                requested_qty,
                entry_limit_price,
                filled_qty,
                avg_entry_price,
                filled_at,
                take_profit_price,
                take_profit_client_order_id,
                take_profit_order_id,
                exit_deadline,
                created_at,
                updated_at
            )
            VALUES (
                ?, ?, ?, ?, ?, ?, ?,
                ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?
            )
            """,
            (
                key,
                payload["date"],
                trade["story_id"],
                trade["ticker"],
                trade["direction"],
                status,
                TRADE_NOTIONAL_USD,
                entry_client_order_id,
                entry_order_id,
                requested_qty,
                entry_limit_price,
                filled_qty,
                avg_entry_price,
                filled_at,
                take_profit_price,
                take_profit_client_order_id,
                take_profit_order_id,
                exit_deadline,
                utc_now(),
                utc_now(),
            ),
        )

    conn.commit()

def submit_entry(
    ticker: str,
    direction: str,
    qty: int,
    limit_price: float,
    client_order_id: str,
):

    side = (
        OrderSide.BUY
        if direction == "FOR"
        else OrderSide.SELL
    )

    request = LimitOrderRequest(
        symbol=ticker,
        qty=qty,
        side=side,
        time_in_force=TimeInForce.DAY,
        limit_price=limit_price,
        extended_hours=False,
        client_order_id=(
            client_order_id
        ),
    )

    return (
        trading_client
        .submit_order(
            order_data=request
        )
    )


def manage_entry_order(
    order,
    ticker: str,
    direction: str,
    increment: Decimal,
):

    started = time.monotonic()

    next_reprice = (
        started
        + ENTRY_REPRICE_MINUTES
        * 60
    )

    max_wait_seconds = (
        MAX_ENTRY_WAIT_MINUTES
        * 60
    )

    while True:

        current = (
            trading_client
            .get_order_by_id(
                order.id
            )
        )

        logger.info(
            "Entry %s status=%s "
            "filled=%s/%s "
            "limit=%s",
            ticker,
            current.status,
            current.filled_qty,
            current.qty,
            current.limit_price,
        )

        if (
            current.status
            == OrderStatus.FILLED
        ):
            return current

        if (
            is_terminal_status(
                current.status
            )
            and current.status
            != OrderStatus.FILLED
        ):
            return current

        elapsed = (
            time.monotonic()
            - started
        )

        if (
            elapsed
            >= max_wait_seconds
        ):
            logger.info(
                "Maximum entry wait reached."
            )

            break

        clock = (
            trading_client
            .get_clock()
        )

        if not clock.is_open:
            logger.info(
                "Market closed before "
                "entry completed."
            )

            break

        if (
            time.monotonic()
            >= next_reprice
        ):
            quote = latest_quote(
                ticker
            )

            if (
                quote[
                    "spread_bps"
                ]
                <= MAX_SPREAD_BPS
            ):
                new_limit = (
                    entry_limit_from_quote(
                        quote,
                        direction,
                        increment,
                    )
                )

                current_limit = float(
                    current.limit_price
                    or 0
                )

                if (
                    new_limit
                    != current_limit
                ):
                    logger.info(
                        "Repricing %s entry "
                        "%.4f -> %.4f",
                        ticker,
                        current_limit,
                        new_limit,
                    )

                    current = (
                        trading_client
                        .replace_order_by_id(
                            current.id,
                            ReplaceOrderRequest(
                                limit_price=(
                                    new_limit
                                )
                            ),
                        )
                    )

                    # Replacement creates/returns the
                    # currently active order object.
                    order = current

            else:
                logger.warning(
                    "%s spread %.1f bps exceeds "
                    "limit; not repricing right now.",
                    ticker,
                    quote[
                        "spread_bps"
                    ],
                )

            next_reprice = (
                time.monotonic()
                + ENTRY_REPRICE_MINUTES
                * 60
            )

        time.sleep(
            ENTRY_POLL_SECONDS
        )

    # Cancel remaining quantity.
    current = (
        trading_client
        .get_order_by_id(
            order.id
        )
    )

    if is_order_open(
        current.status
    ):
        try:
            trading_client.cancel_order_by_id(
                current.id
            )

            time.sleep(2)

        except Exception:
            logger.exception(
                "Entry cancellation returned an error."
            )

    return (
        trading_client
        .get_order_by_id(
            order.id
        )
    )


# ============================================================
# TAKE PROFIT
# ============================================================

def calculate_take_profit(
    avg_entry_price: float,
    direction: str,
    increment: Decimal,
) -> float:

    if direction == "FOR":
        raw = (
            avg_entry_price
            * (
                1
                + TAKE_PROFIT_PCT
            )
        )

        # Long needs sell price >= target.
        return round_to_increment(
            raw,
            increment,
            up=True,
        )

    raw = (
        avg_entry_price
        * (
            1
            - TAKE_PROFIT_PCT
        )
    )

    # Short needs buy price <= target.
    return round_to_increment(
        raw,
        increment,
        up=False,
    )


def submit_take_profit(
    ticker: str,
    direction: str,
    qty: int,
    take_profit_price: float,
    client_order_id: str,
):

    side = (
        OrderSide.SELL
        if direction == "FOR"
        else OrderSide.BUY
    )

    request = LimitOrderRequest(
        symbol=ticker,
        qty=qty,
        side=side,
        time_in_force=TimeInForce.GTC,
        limit_price=(
            take_profit_price
        ),
        extended_hours=False,
        client_order_id=(
            client_order_id
        ),
    )

    return (
        trading_client
        .submit_order(
            order_data=request
        )
    )

# OUTPUT

def write_execution_output(
    brody_payload: dict,
    *,
    status: str,
    details: dict | None = None,
) -> None:

    output = {
        "date": brody_payload[
            "date"
        ],
        "generated_at": utc_now(),
        "paper": True,
        "brody_decision": (
            brody_payload[
                "decision"
            ]
        ),
        "status": status,
        "details": (
            details
            or {}
        ),
    }

    dated_path = (
        DATA_DIR
        / (
            f"{brody_payload['date']}"
            ".alpaca-execution.json"
        )
    )

    dated_path.write_text(
        json.dumps(
            output,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

    EXECUTION_LATEST_PATH.write_text(
        json.dumps(
            output,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

# MAIN
# ============================================================

def main():

    logger.info(
        "Starting Alpaca PAPER executor"
    )

    brody = load_brody()

    if (
        brody["decision"]
        == "NO_TRADE"
    ):
        logger.info(
            "Brody selected NO_TRADE. "
            "No order will be submitted."
        )

        write_execution_output(
            brody,
            status="NO_TRADE",
        )

        return

    trade = brody.get(
        "selected_trade"
    )

    if not trade:
        raise RuntimeError(
            "Brody says TRADE but selected_trade is missing."
        )

    ticker = (
        trade[
            "ticker"
        ].upper()
    )

    if ticker == "STRA":
        raise RuntimeError(
            "Trading STRA is prohibited."
        )

    direction = trade[
        "direction"
    ]

    if direction not in {
        "FOR",
        "AGAINST",
    }:
        raise RuntimeError(
            f"Unsupported direction: {direction}"
        )

    key = decision_key(
        brody
    )

    conn = connect_db()

    try:
        existing = conn.execute(
            """
            SELECT *
            FROM executions
            WHERE decision_key = ?
            """,
            (key,),
        ).fetchone()

        if (
            existing
            and existing["status"]
            in {
                "ENTRY_SUBMITTED",
                "PARTIALLY_FILLED",
                "POSITION_OPEN",
                "TAKE_PROFIT_OPEN",
                "CLOSED",
            }
        ):
            logger.info(
                "Decision already executed or active. "

def analyze(
    payload: dict,
) -> dict:

    candidates = payload.get(
        "candidates",
        [],
    )

    if not candidates:
        return {
            "decision": "NO_TRADE",
            "selected_story_id": None,
            "confidence": 1.0,
            "portfolio_reason": (
                "Gilman supplied no trade candidates."
            ),
            "candidate_assessments": [],
        }

    response = client.responses.create(
        model=BRODY_MODEL,

        reasoning={
            "effort": (
                BRODY_REASONING
            ),
        },

        instructions=(
            BRODY_INSTRUCTIONS
        ),

        input=json.dumps(
            {
                "date": payload["date"],
                "benchmark": (
                    payload.get(
                        "benchmark"
                    )
                ),
                "candidates": (
                    candidates
                ),
            },
            ensure_ascii=False,
        ),

        text={
            "format": {
                "type": "json_schema",
                "name": "brody_decision",
                "strict": True,
                "schema": BRODY_SCHEMA,
            }
        },

        max_output_tokens=16000,

        store=False,
    )

    if not response.output_text:
        raise RuntimeError(
            "Brody returned no output."
        )

    return json.loads(
        response.output_text
    )

def validate_result(
    input_payload: dict,
    result: dict,
) -> dict:

    candidates = input_payload.get(
        "candidates",
        [],
    )

    canonical = {
        candidate[
            "gilman_trade"
        ][
            "story_id"
        ]: candidate
        for candidate
        in candidates
    }

    assessments = result[
        "candidate_assessments"
    ]

    returned_ids = {
        assessment[
            "story_id"
        ]
        for assessment
        in assessments
    }

    expected_ids = set(
        canonical
    )

    if returned_ids != expected_ids:
        raise RuntimeError(
            "Brody did not return exactly one "
            "assessment for every candidate."
        )

    approved = []

    for assessment in assessments:
        story_id = assessment[
            "story_id"
        ]

        original_trade = canonical[
            story_id
        ][
            "gilman_trade"
        ]

        if (
            assessment["ticker"]
            != original_trade["ticker"]
        ):
            raise RuntimeError(
                f"Brody changed ticker for {story_id}."
            )

        if (
            assessment["direction"]
            != original_trade["direction"]
        ):
            raise RuntimeError(
                f"Brody reversed direction for {story_id}."
            )

        if (
            assessment["verdict"]
            == "APPROVE"
        ):
            approved.append(
                assessment
            )

    if len(approved) > 1:
        raise RuntimeError(
            "Brody approved more than one trade."
        )

    if (
        result["decision"]
        == "NO_TRADE"
    ):
        if (
            result[
                "selected_story_id"
            ]
            is not None
        ):
            raise RuntimeError(
                "NO_TRADE has selected_story_id."
            )

        if approved:
            raise RuntimeError(
                "NO_TRADE contains an APPROVE candidate."
            )

        result[
            "selected_trade"
        ] = None

        return result

    # TRADE
    if len(approved) != 1:
        raise RuntimeError(
            "TRADE must contain exactly one APPROVE."
        )

    selected_story_id = result[
        "selected_story_id"
    ]

    if (
        selected_story_id
        != approved[0]["story_id"]
    ):
        raise RuntimeError(
            "selected_story_id does not match "
            "the approved candidate."
        )

    selected_candidate = canonical[
        selected_story_id
    ]

    # IMPORTANT:
    # Copy the canonical Gilman object from our own input rather
    # than trusting the LLM to regenerate ticker/direction/thesis.
    result["selected_trade"] = (
        selected_candidate[
            "gilman_trade"
        ]
    )

    result["selected_market_context"] = {
        key: value
        for key, value
        in selected_candidate.items()
        if key != "gilman_trade"
    }

    return result

def write_output(
    source_payload: dict,
    result: dict,
) -> None:

    date_str = source_payload[
        "date"
    ]

    output = {
        "date": date_str,
        "generated_at": utc_now(),
        "model": BRODY_MODEL,
        "reasoning_effort": (
            BRODY_REASONING
        ),
        "candidate_count": len(
            source_payload.get(
                "candidates",
                [],
            )
        ),
        **result,
    }

    dated_path = (
        DATA_DIR
        / f"{date_str}.brody.json"
    )

    dated_path.write_text(
        json.dumps(
            output,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

    shutil.copyfile(
        dated_path,
        BRODY_LATEST_PATH,
    )

    logger.info(
        "Wrote %s",
        dated_path,
    )

    logger.info(
        "Wrote %s",
        BRODY_LATEST_PATH,
    )

    if (
        output["decision"]
        == "TRADE"
    ):
        trade = output[
            "selected_trade"
        ]

        logger.info(
            "BRODY APPROVED: %s %s | confidence=%.2f",
            trade["direction"],
            trade["ticker"],
            output["confidence"],
        )

    else:
        logger.info(
            "BRODY: NO_TRADE"
        )


# ============================================================
# MAIN
# ============================================================

def main():
    logger.info(
        "Starting Brody"
    )

    payload = load_input()

    logger.info(
        "Evaluating %d Gilman candidates",
        len(
            payload.get(
                "candidates",
                [],
            )
        ),
    )

    result = analyze(
        payload
    )

    result = validate_result(
        payload,
        result,
    )

    write_output(
        payload,
        result,
    )

    logger.info(
        "Brody complete"
    )


if __name__ == "__main__":
    main()

)

CONNECTIVITY_TIMEOUT_SECONDS = 10


# This stage is stateful: it can submit brokerage orders.
#
# Do NOT generically rerun it just because the process returned
# an error. alpaca_execute.py should handle order reconciliation
# and idempotency internally.
NON_RETRYABLE_STAGES = {
    "scripts.alpaca_execute",
}


# ============================================================
# CONNECTIVITY
# ============================================================

def has_internet() -> bool:
    """
    Test actual HTTPS connectivity.

    Google's generate_204 endpoint normally returns exactly 204.

    A coffee-shop captive portal will often redirect this request
    instead, which we deliberately treat as "not online yet".
    """

    try:
        response = requests.get(
            CONNECTIVITY_URL,
            timeout=CONNECTIVITY_TIMEOUT_SECONDS,
            allow_redirects=False,
        )

        return response.status_code == 204

    except requests.RequestException:
        return False


def wait_for_internet() -> None:
    for attempt in range(
        1,
        MAX_CONNECTIVITY_ATTEMPTS + 1,
    ):
        if has_internet():
            logger.info(
                "✓ Internet connection is available"
            )
            return

        logger.warning(
            "No usable internet connection yet "
            "(attempt %d/%d)",
            attempt,
            MAX_CONNECTIVITY_ATTEMPTS,
        )

        if (
            attempt
            >= MAX_CONNECTIVITY_ATTEMPTS
        ):
            raise RuntimeError(
                "No usable internet connection "
                "after all connectivity attempts."
            )

        logger.info(
            "Checking again in %d minutes...",
            CONNECTIVITY_RETRY_SECONDS // 60,
        )

        time.sleep(
            CONNECTIVITY_RETRY_SECONDS
        )

def load_json(
    path: Path,
) -> dict:

    if not path.exists():
        raise RuntimeError(
            f"Expected output does not exist: {path}"
        )

    with path.open(
        "r",
        encoding="utf-8",
    ) as f:
        return json.load(f)

def run_stage_once(
    module: str,
) -> None:

    logger.info(
        "============================================================"
    )

    logger.info(
        "Starting stage: %s",
        module,
    )

    subprocess.run(
        [
            sys.executable,
            "-u",
            "-m",
            module,
        ],
        cwd=ROOT,
        check=True,
    )

    logger.info(
        "Completed stage: %s",
        module,
    )


def run_stage(
    module: str,
) -> None:
    """
    Run one stage.

    Normal data/LLM stages can retry after transient failures.

    alpaca_execute is deliberately run only once because a failed
    brokerage request may have succeeded remotely even if our local
    process did not receive the response.
    """

    if module in NON_RETRYABLE_STAGES:
        wait_for_internet()

        run_stage_once(
            module
        )

        return

    for attempt in range(
        1,
        MAX_STAGE_ATTEMPTS + 1,
    ):
        try:
            wait_for_internet()

            logger.info(
                "%s attempt %d/%d",
                module,
                attempt,
                MAX_STAGE_ATTEMPTS,
            )

            run_stage_once(
                module
            )

            return

        except subprocess.CalledProcessError as exc:
            logger.error(
                "%s failed on attempt %d/%d "
                "with exit code %s",
                module,
                attempt,
                MAX_STAGE_ATTEMPTS,
                exc.returncode,
            )

            if (
                attempt
                >= MAX_STAGE_ATTEMPTS
            ):
                raise

            logger.info(
                "Retrying %s in %d minutes...",
                module,
                STAGE_RETRY_SECONDS // 60,
            )

            time.sleep(
                STAGE_RETRY_SECONDS
            )

def validate_gilman() -> None:
    data = load_json(
        DATA_DIR / "gilman_latest.json"
    )

    story_count = data.get(
        "story_count"
    )

    trade_count = data.get(
        "trade_count"
    )

    no_trade_count = data.get(
        "no_trade_count"
    )

    if story_count != 100:
        raise RuntimeError(
            f"Gilman analyzed "
            f"{story_count} stories; expected 100."
        )

    if not isinstance(
        trade_count,
        int,
    ):
        raise RuntimeError(
            "Gilman output has no valid trade_count."
        )

    if not isinstance(
        no_trade_count,
        int,
    ):
        raise RuntimeError(
            "Gilman output has no valid no_trade_count."
        )

    if (
        trade_count
        + no_trade_count
        != story_count
    ):
        raise RuntimeError(
            "Gilman counts do not add up."
        )

    trades = load_json(
        DATA_DIR
        / "gilman_trades_latest.json"
    )

    if (
        trades.get("trade_count")
        != trade_count
    ):
        raise RuntimeError(
            "Gilman output files disagree "
            "about trade_count."
        )

    logger.info(
        "✓ GILMAN: %d TRADE | %d NO_TRADE",
        trade_count,
        no_trade_count,
    )


def validate_market_context() -> None:
    gilman = load_json(
        DATA_DIR
        / "gilman_trades_latest.json"
    )

    market = load_json(

    expected = gilman.get(

            "market_context candidate_count "
            "does not match candidates array."

    logger.info(
        "✓ MARKET CONTEXT: %d candidates",
        actual,
    )


def validate_brody() -> None:
    market = load_json(
        DATA_DIR
        / "market_context_latest.json"
    )

    brody = load_json(
        DATA_DIR
        / "brody_latest.json"
    )

    expected_candidates = (
        market.get(
            "candidate_count",
            0,
        )
    )

    actual_candidates = (
        brody.get(
            "candidate_count"
        )
    )

    if (
        actual_candidates
        != expected_candidates
    ):
        raise RuntimeError(
            "Brody candidate count does not "
            "match market_context."
        )

    decision = brody.get(
        "decision"
    )

    if decision not in {
        "TRADE",
        "NO_TRADE",
    }:
        raise RuntimeError(
            f"Invalid Brody decision: {decision}"
        )

    assessments = brody.get(
        "candidate_assessments",
        [],
    )

    if (
        len(assessments)
        != expected_candidates
    ):
        raise RuntimeError(
            "Brody did not assess every candidate."
        )

    approved = [
        item
        for item in assessments
        if item.get(
            "verdict"
        ) == "APPROVE"
    ]

    if decision == "NO_TRADE":
        if approved:
            raise RuntimeError(
                "Brody says NO_TRADE but "
                "approved a candidate."
            )

        if (
            brody.get(
                "selected_trade"
            )
            is not None
        ):
            raise RuntimeError(
                "Brody says NO_TRADE but "
                "selected_trade is populated."
            )

        logger.info(
            "✓ BRODY: NO_TRADE"
        )

        return

    # TRADE
    if len(approved) != 1:
        raise RuntimeError(
            "Brody TRADE must contain "
            "exactly one approved candidate."
        )

    selected = brody.get(
        "selected_trade"
    )

    if not selected:
        raise RuntimeError(
            "Brody says TRADE but "
            "selected_trade is missing."
        )

    logger.info(
        "✓ BRODY: APPROVED %s %s",
        selected["direction"],
        selected["ticker"],
    )


def validate_alpaca_execute() -> None:
    brody = load_json(
        DATA_DIR
        / "brody_latest.json"
    )

    execution = load_json(
        DATA_DIR
        / "alpaca_execution_latest.json"
    )

    status = execution.get(
        "status"
    )

    allowed_statuses = {
        "NO_TRADE",
        "NO_FILL",
        "POSITION_OPEN",
        "ALREADY_EXECUTED",
    }

    if status not in allowed_statuses:
        raise RuntimeError(
            f"Unexpected Alpaca execution "
            f"status: {status}"
        )

    if (
        brody["decision"]
        == "NO_TRADE"
        and status != "NO_TRADE"
    ):
        raise RuntimeError(
            "Brody said NO_TRADE but Alpaca "
            "execution did something else."
        )

    if (
        brody["decision"]
        == "TRADE"
        and status == "NO_TRADE"
    ):
        raise RuntimeError(
            "Brody approved a trade but Alpaca "
            "reported NO_TRADE."
        )

    if status == "POSITION_OPEN":
        details = execution.get(
            "details",
            {},
        )

        logger.info(
            "✓ ALPACA: POSITION OPEN | "
            "%s %s | qty=%s | entry=%.4f | "
            "target=%.4f",
            details.get(
                "direction"
            ),
            details.get(
                "ticker"
            ),
            details.get(
                "qty"
            ),
            details.get(
                "avg_entry_price"
            ),
            details.get(
                "take_profit_price"
            ),
        )

    elif status == "NO_FILL":
        logger.info(
            "✓ ALPACA: approved trade did not "
            "fill within execution window"
        )

    elif status == "ALREADY_EXECUTED":
        logger.info(
            "✓ ALPACA: trade was already executed; "
            "duplicate prevented"
        )

    else:
        logger.info(
            "✓ ALPACA: NO_TRADE"
        )


VALIDATORS = {
    "scripts.news_collect": (
        validate_news_collect
    ),
    "scripts.event_memory": (
        validate_event_memory
    ),
    "scripts.gilman": (
        validate_gilman
    ),
    "scripts.market_context": (
        validate_market_context
    ),
    "scripts.brody": (
        validate_brody
    ),
    "scripts.alpaca_execute": (
        validate_alpaca_execute
    ),
}


# ============================================================
# PIPELINE
# ============================================================

def main():
    logger.info(
        "============================================================"
    )

    logger.info(
        "Starting daily news → Gilman → "
        "Brody → Alpaca pipeline"
    )

    # Important for the "opened laptop at coffee shop" case.
    wait_for_internet()

    for module in STAGES:

        run_stage(
            module
        )

        validator = VALIDATORS.get(
            module
        )

        if validator:
            # Validation failures deliberately DO NOT retry.
            #
            # If a script exited successfully but produced bad or
            # inconsistent output, that probably isn't a Wi-Fi
            # problem. Fail loudly instead of paying to rerun it
            # repeatedly.
            validator()

    logger.info(
        "============================================================"
    )

    logger.info(
        "DAILY PIPELINE SUCCESS"
    )


if __name__ == "__main__":
    main()

def utc_now() -> str:
    return datetime.now(
        timezone.utc
    ).isoformat()


def normalized_title(value: str) -> str:
    value = value.casefold()

    value = re.sub(
        r"[^a-z0-9]+",
        " ",
        value,
    )

    return " ".join(
        value.split()
    )


def make_story_id(
    story_date: str,
    story: dict,
) -> str:
    """
    Stable ID based on date + canonical headline.

    We do not use rank because rank can change if the collector
    is rerun on the same day.
    """

    if story.get("story_id"):
        return story["story_id"]

    digest = hashlib.sha1(
        normalized_title(
            story["title"]
        ).encode("utf-8")
    ).hexdigest()[:12]

    compact_date = story_date.replace(
        "-",
        "",
    )

    return f"story_{compact_date}_{digest}"

def embed_texts(
    texts: list[str],
) -> np.ndarray:

    if not texts:
        return np.empty(
            (0, EMBEDDING_DIMENSIONS),
            dtype=np.float32,
        )

    response = client.embeddings.create(
        model=EMBEDDING_MODEL,
        input=texts,
        dimensions=EMBEDDING_DIMENSIONS,
    )

    ordered = sorted(
        response.data,
        key=lambda item: item.index,
    )

    matrix = np.asarray(
        [
            item.embedding
            for item in ordered
        ],
        dtype=np.float32,
    )

    # Normalize so dot product = cosine similarity.
    norms = np.linalg.norm(
        matrix,
        axis=1,
        keepdims=True,
    )

    norms[norms == 0] = 1

    return matrix / norms


def vector_to_blob(
    vector: np.ndarray,
) -> bytes:
    return np.asarray(
        vector,
        dtype=np.float32,
    ).tobytes()


def blob_to_vector(
    blob: bytes,
) -> np.ndarray:
    vector = np.frombuffer(
        blob,
        dtype=np.float32,
    ).copy()

    norm = np.linalg.norm(vector)

    if norm:
        vector /= norm

    return vector

STOP_WORDS = {
    "the",
    "a",
    "an",
    "and",
    "or",
    "of",
    "to",
    "in",
    "on",
    "for",
    "at",
    "by",
    "with",
    "from",
    "after",
    "as",
    "is",
    "are",
    "was",
    "were",
    "be",
    "has",
    "have",
    "had",
    "says",
    "say",
    "amid",
    "over",
    "into",
    "new",
}


def lexical_tokens(
    value: str,
) -> set[str]:

    words = re.findall(
        r"[a-z0-9][a-z0-9.-]+",
        value.casefold(),
    )

    return {
        word
        for word in words
        if (
            len(word) >= 3
            and word not in STOP_WORDS
        )
    }


def lexical_similarity(
    query: str,
    document: str,
) -> float:

    q = lexical_tokens(query)
    d = lexical_tokens(document)

    if not q or not d:
        return 0.0

    intersection = len(
        q.intersection(d)
    )

    # More useful than pure Jaccard for proper nouns because
    # we care how much of the query is explained by the event.
    return intersection / len(q)

def load_event_index(
    conn: sqlite3.Connection,
) -> list[dict]:

    rows = conn.execute(
        """
        SELECT
            event_id,
            canonical_title,
            summary,
            first_seen,
            last_seen,
            embedding
        FROM events
        """
    ).fetchall()

    events = []

    for row in rows:
        vector = blob_to_vector(
            row["embedding"]
        )

        # Protect against changing EMBEDDING_DIMENSIONS later.
        if len(vector) != EMBEDDING_DIMENSIONS:
            continue

        events.append(
            {
                "event_id": row["event_id"],
                "canonical_title": row["canonical_title"],
                "summary": row["summary"],
                "first_seen": row["first_seen"],
                "last_seen": row["last_seen"],
                "vector": vector,
            }
        )

    return events


def get_recent_event_timeline(
    conn: sqlite3.Connection,
    event_id: str,
    limit: int = TIMELINE_ITEMS_PER_EVENT,
) -> list[dict]:

    rows = conn.execute(
        """
        SELECT
            story_id,
            story_date,
            story_rank,
            title,
            description,
            update_type,
            novelty
        FROM stories
        WHERE event_id = ?
        ORDER BY story_date DESC, story_rank ASC
        LIMIT ?
        """,
        (
            event_id,
            limit,
        ),
    ).fetchall()

    timeline = [
        {
            "story_id": row["story_id"],
            "date": row["story_date"],
            "title": row["title"],
            "description": row["description"],
            "update_type": row["update_type"],
            "novelty": row["novelty"],
        }
        for row in rows
    ]

    timeline.reverse()

    return timeline


def retrieve_event_candidates(
    conn: sqlite3.Connection,
    event_index: list[dict],
    story: dict,
    story_vector: np.ndarray,
) -> list[dict]:

    if not event_index:
        return []

    query_text = (
        f"{story['title']} "
        f"{story.get('description', '')}"
    )

    scored = []

    for event in event_index:
        semantic = float(
            np.dot(
                story_vector,
                event["vector"],
            )
        )

        lexical = lexical_similarity(
            query_text,
            (
                f"{event['canonical_title']} "
                f"{event['summary']}"
            ),
        )

        # Semantic retrieval carries most of the weight.
        combined = (
            0.87 * semantic
            + 0.13 * lexical
        )

        scored.append(
            (
                combined,
                semantic,
                lexical,
                event,
            )
        )

    scored.sort(
        key=lambda item: item[0],
        reverse=True,
    )

    results = []

    for (
        combined,
        semantic,
        lexical,
        event,
    ) in scored[:MAX_EVENT_CANDIDATES]:

        results.append(
            {
                "event_id": event["event_id"],
                "canonical_title": (
                    event["canonical_title"]
                ),
                "summary": event["summary"],
                "first_seen": event["first_seen"],
                "last_seen": event["last_seen"],
                "retrieval_score": round(
                    combined,
                    4,
                ),
                "semantic_similarity": round(
                    semantic,
                    4,
                ),
                "lexical_similarity": round(
                    lexical,
                    4,
                ),
                "recent_timeline": (
                    get_recent_event_timeline(
                        conn,
                        event["event_id"],
                    )
                ),
            }
        )

    return results

EVENT_MEMORY_INSTRUCTIONS = """
You maintain a persistent temporal memory of major news events.

You receive up to 10 CURRENT STORIES. For each current story you also
receive zero or more CANDIDATE HISTORICAL EVENTS retrieved from a
database.

Your job is to decide how the current story relates to history.

Be conservative about SAME_EVENT.

Two stories are SAME_EVENT when they concern the same persistent
real-world occurrence, case, transaction, investigation, conflict,
policy action, disaster, corporate situation, legal proceeding, etc.

Merely sharing:
- a person,
- company,
- country,
- industry,
- war,
- broad political issue,
- product,
or theme

does NOT make two stories the same event.

Examples:

Original:
"Boeing loses $20 billion lawsuit"

Later:
"Boeing CEO discusses $20 billion lawsuit"

=> SAME_EVENT
=> likely REACTION_OR_COMMENTARY
=> novelty should be relatively low.

Original:
"Russia cuts gas supplies to Europe"

Later:
"European fertilizer plants shut as gas prices surge"

=> not SAME_EVENT.
The second may be CONSEQUENCE_OF the first.

For every current story:

1. Select matched_event_id ONLY from its supplied candidates, or null.
2. Decide relationship.
3. Classify update_type.
4. Estimate novelty from 0 to 1.
5. Produce a compact canonical event title.
6. Produce an UPDATED rolling event summary.
7. Write a historical_note explaining what today's headline adds
   relative to what was already known.
8. Identify any especially important prior story IDs.

If SAME_EVENT:
- matched_event_id must be populated.
- event_summary must update that existing event summary with the new
  information while preserving important prior chronology.

If the story deserves its own new event:
- relationship may be UNRELATED, CAUSE_OF, CONSEQUENCE_OF, or
  RELATED_CONTEXT.
- canonical_event_title and event_summary describe the NEW event.
- If there is an important historical relationship, matched_event_id
  points to that older event.

Do not overstate novelty.
A dramatic headline can still contain little new information.
"""

def link_story_batch(
    batch_payload: list[dict],
) -> list[dict]:

    response = client.responses.create(
        model=EVENT_MODEL,
        reasoning={
            "effort": EVENT_REASONING,
        },
        instructions=EVENT_MEMORY_INSTRUCTIONS,
        input=json.dumps(
            {
                "current_stories": batch_payload,
            },
            ensure_ascii=False,
        ),
        text={
            "format": {
                "type": "json_schema",
                "name": "event_memory_decisions",
                "strict": True,
                "schema": EVENT_LINK_SCHEMA,
            }
        },
        max_output_tokens=12000,
        store=False,
    )

    if not response.output_text:
        raise RuntimeError(
            "Event Memory returned no output."
        )

    result = json.loads(
        response.output_text
    )

    return result["decisions"]


def create_event_id() -> str:
    return (
        "evt_"
        + uuid.uuid4().hex[:16]
    )


def refresh_event_embeddings(
    conn: sqlite3.Connection,
    event_ids: set[str],
) -> None:

    if not event_ids:
        return

    rows = []

    for event_id in event_ids:
        row = conn.execute(
            """
            SELECT
                event_id,
                canonical_title,
                summary
            FROM events
            WHERE event_id = ?
            """,
            (event_id,),
        ).fetchone()

        if row:
            rows.append(row)

    if not rows:
        return

    vectors = embed_texts(
        [
            event_embedding_text(
                row["canonical_title"],
                row["summary"],
            )
            for row in rows
        ]
    )

    for row, vector in zip(
        rows,
        vectors,
    ):
        conn.execute(
            """
            UPDATE events
            SET
                embedding = ?,
                updated_at = ?
            WHERE event_id = ?
            """,
            (
                vector_to_blob(vector),
                utc_now(),
                row["event_id"],
            ),
        )


def apply_decision(
    conn: sqlite3.Connection,
    story_date: str,
    story: dict,
    story_id: str,
    story_vector: np.ndarray,
    candidates: list[dict],
    decision: dict,
) -> str:

    supplied_event_ids = {
        candidate["event_id"]
        for candidate in candidates
    }

    matched_event_id = (
        decision["matched_event_id"]
    )

    if (
        matched_event_id
        and matched_event_id
        not in supplied_event_ids
    ):
        logger.warning(
            "Model returned event not supplied for %s: %s",
            story_id,
            matched_event_id,
        )

        matched_event_id = None

    relationship = decision[
        "relationship"
    ]

    # --------------------------------------------------------
    # SAME EVENT
    # --------------------------------------------------------

    if (
        relationship == "SAME_EVENT"
        and matched_event_id
    ):
        event_id = matched_event_id

        existing = conn.execute(
            """
            SELECT first_seen
            FROM events
            WHERE event_id = ?
            """,
            (event_id,),
        ).fetchone()

        if not existing:
            raise RuntimeError(
                f"Missing event {event_id}"
            )

        conn.execute(
            """
            UPDATE events
            SET
                canonical_title = ?,
                summary = ?,
                last_seen = ?,
                updated_at = ?
            WHERE event_id = ?
            """,
            (
                decision[
                    "canonical_event_title"
                ],
                decision[
                    "event_summary"
                ],
                story_date,
                utc_now(),
                event_id,
            ),
        )

        related_event_id = None

    # --------------------------------------------------------
    # NEW EVENT
    # --------------------------------------------------------

    else:
        event_id = create_event_id()

        related_event_id = (
            matched_event_id
            if (
                matched_event_id
                and relationship
                in {
                    "CAUSE_OF",
                    "CONSEQUENCE_OF",
                    "RELATED_CONTEXT",
                }
            )
            else None
        )

        conn.execute(
            """
            INSERT INTO events (
                event_id,
                canonical_title,
                summary,
                first_seen,
                last_seen,
                embedding,
                created_at,
                updated_at
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            """,
            (
                event_id,
                decision[
                    "canonical_event_title"
                ],
                decision[
                    "event_summary"
                ],
                story_date,
                story_date,
                vector_to_blob(
                    story_vector
                ),
                utc_now(),
                utc_now(),
            ),
        )

        if related_event_id:
            conn.execute(
                """
                INSERT OR IGNORE INTO
                event_relationships (
                    from_event_id,
                    to_event_id,
                    relationship,
                    created_at
                )
                VALUES (?, ?, ?, ?)
                """,
                (
                    event_id,
                    related_event_id,
                    relationship,
                    utc_now(),
                ),
            )

    conn.execute(
        """
        INSERT INTO stories (
            story_id,
            story_date,
            story_rank,
            title,
            description,
            event_id,
            update_type,
            relationship,
            novelty,
            historical_note,
            related_event_id,
            key_prior_story_ids_json,
            created_at
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """,
        (
            story_id,
            story_date,
            story["rank"],
            story["title"],
            story.get(
                "description",
                "",
            ),
            event_id,
            decision["update_type"],
            relationship,
            float(
                decision["novelty"]
            ),
            decision[
                "historical_note"
            ],
            related_event_id,
            json.dumps(
                decision[
                    "key_prior_story_ids"
                ]
            ),
            utc_now(),
        ),
    )

    return event_id

def process_news_payload(
    conn: sqlite3.Connection,
    payload: dict,
) -> None:

    story_date = payload["date"]

    pending = []

    for story in payload["stories"]:
        story_id = make_story_id(
            story_date,
            story,
        )

        exists = conn.execute(
            """
            SELECT 1
            FROM stories
            WHERE story_id = ?
            """,
            (story_id,),
        ).fetchone()

        if exists:
            continue

        pending.append(
            {
                "story_id": story_id,
                "story": story,
            }
        )

    logger.info(
        "%s: %d unprocessed stories",
        story_date,
        len(pending),
    )

    for start in range(
        0,
        len(pending),
        BATCH_SIZE,
    ):
        batch = pending[
            start:start + BATCH_SIZE
        ]

        logger.info(
            "Event Memory: stories %d-%d of %d for %s",
            start + 1,
            start + len(batch),
            len(pending),
            story_date,
        )

        story_vectors = embed_texts(
            [
                story_embedding_text(
                    item["story"]
                )
                for item in batch
            ]
        )

        # Refresh between batches so the next ten stories can
        # retrieve events created by previous batches.
        event_index = load_event_index(
            conn
        )

        batch_payload = []
        candidate_map = {}

        for item, vector in zip(
            batch,
            story_vectors,
        ):
            candidates = (
                retrieve_event_candidates(
                    conn,
                    event_index,
                    item["story"],
                    vector,
                )
            )

            candidate_map[
                item["story_id"]
            ] = candidates

            batch_payload.append(
                {
                    "story_id": (
                        item["story_id"]
                    ),
                    "date": story_date,
                    "rank": (
                        item["story"]["rank"]
                    ),
                    "title": (
                        item["story"]["title"]
                    ),
                    "description": (
                        item["story"].get(
                            "description",
                            "",
                        )
                    ),
                    "candidate_historical_events": (
                        candidates
                    ),
                }
            )

        decisions = link_story_batch(
            batch_payload
        )

        decisions_by_id = {
            decision["story_id"]: decision
            for decision in decisions
        }

        expected_ids = {
            item["story_id"]
            for item in batch
        }

        if set(
            decisions_by_id
        ) != expected_ids:
            raise RuntimeError(
                "Event Memory did not return exactly "
                "one decision for every story."
            )

        touched_event_ids = set()

        for (
            item,
            vector,
        ) in zip(
            batch,
            story_vectors,
        ):
            story_id = item[
                "story_id"
            ]

            decision = (
                decisions_by_id[
                    story_id
                ]
            )

            event_id = apply_decision(
                conn=conn,
                story_date=story_date,
                story=item["story"],
                story_id=story_id,
                story_vector=vector,
                candidates=(
                    candidate_map[
                        story_id
                    ]
                ),
                decision=decision,
            )

            touched_event_ids.add(
                event_id
            )

        refresh_event_embeddings(
            conn,
            touched_event_ids,
        )

        conn.commit()

DATE_FILE_RE = re.compile(

def dated_news_files() -> list[Path]:
    files = [
        path
        for path in DATA_DIR.iterdir()
        if (
            path.is_file()
            and DATE_FILE_RE.match(
                path.name

    return sorted(files)


def load_json(
    path: Path,
) -> dict:

    with path.open(
        "r",
        encoding="utf-8",
    ) as f:
        return json.load(f)

def related_event_info(
    conn: sqlite3.Connection,
    event_id: str | None,
) -> dict | None:

    if not event_id:
        return None

    row = conn.execute(
        """
        SELECT
            event_id,
            canonical_title,
            summary
        FROM events
        WHERE event_id = ?
        """,
        (event_id,),
    ).fetchone()

    if not row:
        return None

    return {
        "event_id": row["event_id"],
        "canonical_title": (
            row["canonical_title"]
        ),
        "summary": row["summary"],
    }


def build_enriched_output(
    conn: sqlite3.Connection,
    latest_payload: dict,
) -> dict:

    story_date = latest_payload[
        "date"
    ]

    enriched_stories = []

    for original_story in latest_payload[
        "stories"
    ]:
        story_id = make_story_id(
            story_date,
            original_story,
        )

        story_row = conn.execute(
            """
            SELECT *
            FROM stories
            WHERE story_id = ?
            """,
            (story_id,),
        ).fetchone()

        if not story_row:
            raise RuntimeError(
                f"Story {story_id} has not "
                "been processed by Event Memory."
            )

        event_row = conn.execute(
            """
            SELECT *
            FROM events
            WHERE event_id = ?
            """,
            (
                story_row[
                    "event_id"
                ],
            ),
        ).fetchone()

        timeline = (
            get_recent_event_timeline(
                conn,
                story_row[
                    "event_id"
                ],
                limit=8,
            )
        )

        enriched_stories.append(
            {
                **original_story,
                "story_id": story_id,
                "event_context": {
                    "event_id": (
                        event_row[
                            "event_id"
                        ]
                    ),
                    "canonical_title": (
                        event_row[
                            "canonical_title"
                        ]
                    ),
                    "first_seen": (
                        event_row[
                            "first_seen"
                        ]
                    ),
                    "last_seen": (
                        event_row[
                            "last_seen"
                        ]
                    ),
                    "rolling_summary": (
                        event_row[
                            "summary"
                        ]
                    ),
                    "update_type": (
                        story_row[
                            "update_type"
                        ]
                    ),
                    "relationship": (
                        story_row[
                            "relationship"
                        ]
                    ),
                    "novelty": (
                        story_row[
                            "novelty"
                        ]
                    ),
                    "historical_note": (
                        story_row[
                            "historical_note"
                        ]
                    ),
                    "related_event": (
                        related_event_info(
                            conn,
                            story_row[
                                "related_event_id"
                            ],
                        )
                    ),
                    "timeline": timeline,
                },
            }
        )

    return {
        **latest_payload,
        "memory_generated_at": (
            utc_now()
        ),
        "stories": enriched_stories,
    }


def write_enriched_latest(
    conn: sqlite3.Connection,
) -> None:

    latest_payload = load_json(
        LATEST_PATH
    )

    enriched = build_enriched_output(
        conn,
        latest_payload,
    )

    date_str = enriched["date"]

    dated_path = (
        DATA_DIR
        / f"{date_str}.enriched.json"
    )

    dated_path.write_text(
        json.dumps(
            enriched,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

    shutil.copyfile(
        dated_path,
        ENRICHED_LATEST_PATH,
    )

    logger.info(
        "Wrote %s",
        dated_path,
    )

    logger.info(
        "Wrote %s",
        ENRICHED_LATEST_PATH,
    )

def main():
    parser = argparse.ArgumentParser()

    parser.add_argument(
        "--backfill",
        action="store_true",
        help=(
            "Process all historical YYYY-MM-DD.json "
            "files before today's latest.json."
        ),
    )

    args = parser.parse_args()

    conn = connect_db()

    try:
        if args.backfill:
            files = dated_news_files()

            logger.info(
                "Backfilling %d dated news files",
                len(files),
            )

            for path in files:
                logger.info(
                    "Processing historical file: %s",
                    path.name,
                )

                payload = load_json(
                    path
                )

                process_news_payload(
                    conn,
                    payload,
                )

        else:
            payload = load_json(
                LATEST_PATH
            )

            process_news_payload(
                conn,
                payload,
            )

        write_enriched_latest(
            conn
        )

    finally:
        conn.close()

    logger.info(
        "Event Memory complete"
    )


if __name__ == "__main__":
    main()

    "Which publicly traded company has meaningful economic exposure
     one or more steps away from the original headline?"

============================================================
CORE PHILOSOPHY
============================================================

Your goal is NOT to generate trades.

NO_TRADE is expected for most news stories.

A clever narrative is not sufficient.

A valid trade requires a concrete and defensible causal mechanism
connecting the event to future economics for a specific publicly
traded company.

Prefer indirect or second-order exposure over obvious headline
exposure, but NEVER choose an indirect company merely because the
connection sounds clever.

One excellent idea is better than ten mediocre ideas.

============================================================
EVENT MEMORY
============================================================

Each story includes an Event Memory object.

USE IT.

A headline may be dramatic while containing almost no new information.

Pay close attention to:

- first_seen
- rolling_summary
- timeline
- update_type
- novelty
- historical_note

Do not mistake commentary, recaps, or reactions to an old event for a
new fundamental shock.

============================================================
WEB RESEARCH
============================================================

You have web search.

Use it selectively.

You do NOT need to search the web for stories that clearly have no
investment implication.

However:

YOU MAY NOT RETURN TRADE UNLESS YOU HAVE USED WEB RESEARCH TO VERIFY
THE RECOMMENDED COMPANY'S ACTUAL ECONOMIC EXPOSURE.

For every proposed trade, verify:

1. The company is publicly traded.
2. The ticker is correct.
3. The company actually has the product, geography, supplier,
   customer, commodity, manufacturing footprint, regulatory exposure,
   or other economic relationship claimed in the thesis.
4. The exposure is economically meaningful enough to make the causal
   chain plausible.

Prefer evidence in roughly this order:

- regulatory filings / regulator sources
- company investor-relations or company primary sources
- highly reputable financial or general news
- credible industry/trade publications

Use at least TWO web sources for a TRADE whenever possible.

At least one evidence item should directly support the claimed company
exposure.

Never invent URLs.

If you cannot verify the exposure, return NO_TRADE.

============================================================
WHAT NOT TO DO YET
============================================================

Another downstream analyst will handle:

- stock-price reactions
- technicals
- valuation
- relative performance
- volume
- factor exposure
- portfolio sizing
- market timing
- earnings dates
- actual execution

Do NOT reject a creative idea merely because you do not have market
price data.

Your task is to discover and verify the economic idea.

"market_connection_hypothesis" should explain WHY the relationship may
be non-obvious, not claim that the market definitely has not priced it.

============================================================
TRADE REQUIREMENTS
============================================================

For TRADE:

- Recommend exactly one company.
- direction is FOR for a long thesis.
- direction is AGAINST for a short thesis.
- Common stock / ADR ideas only.
- No options.
- No crypto.
- No private companies.
- No generic sector calls.
- No ETF unless explicitly necessary; prefer an operating company.
- causal_chain should normally have 3-6 concrete steps.
- State catalysts.
- State risks.
- State falsifiable invalidation conditions.
- Distinguish causal confidence from confidence in the company's
  exposure.
- Confidence is not bravado. Use the full 0-1 range.

Do not create multiple variations of essentially the same thesis.

============================================================
BATCH BEHAVIOR
============================================================

You receive up to 10 stories at a time.

Analyze ALL of them.

Compare them against each other.

Spend research effort only where there is plausible investment
relevance.

Return exactly one analysis object for every supplied story_id.
"""

def normalize_url(
    value: str,
) -> str:

    if not value:
        return ""

    try:
        parts = urlsplit(
            value.strip()
        )

        return urlunsplit(
            (
                parts.scheme.casefold(),
                parts.netloc.casefold(),
                parts.path.rstrip("/"),
                "",
                "",
            )
        )

    except Exception:
        return value.strip()


def extract_web_sources(
    response,
) -> list[dict]:
    """
    We request web_search_call.action.sources from the API.

    Keep the complete consulted-source list as an audit trail.
    """

    data = response.model_dump()

    found = {}

    for item in data.get(
        "output",
        [],
    ):
        if (
            item.get("type")
            != "web_search_call"
        ):
            continue

        action = (
            item.get("action")
            or {}
        )

        for source in action.get(
            "sources",
            [],
        ):
            if not isinstance(
                source,
                dict,
            ):
                continue

            url = source.get(
                "url"
            )

            if not url:
                continue

            key = normalize_url(
                url
            )

            found[key] = {
                "title": (
                    source.get(
                        "title"
                    )
                    or ""
                ),
                "url": url,
                "type": (
                    source.get(
                        "type"
                    )
                    or ""
                ),
            }

    return list(
        found.values()
    )

def analyze_batch(
    stories: list[dict],

    payload = {

        ]

    response = client.responses.create(

        ),

        tools=[
            {
                "type": "web_search",

            }

        tool_choice="auto",

        include=[

    if not response.output_text:
        raise RuntimeError(

    result = json.loads(
        response.output_text

TICKER_RE = re.compile(
    r"^[A-Z][A-Z0-9.-]{0,11}$"
)


def reject_trade(
    analysis: dict,
    reason: str,
) -> None:

    previous_reason = analysis[
        "reason"
    ]

    analysis["decision"] = (
        "NO_TRADE"
    )

    analysis["reason"] = (
        f"{previous_reason} "
        f"[Automated validation rejected "
        f"the proposed trade: {reason}]"
    )

    analysis["trade"] = None


def validate_trade(
    analysis: dict,
    consulted_sources: list[dict],
) -> None:

    if (
        analysis["decision"]
        != "TRADE"
    ):
        analysis["trade"] = None
        return

    trade = analysis.get(
        "trade"
    )

    if not trade:
        reject_trade(
            analysis,
            "TRADE had no trade object.",
        )
        return

    ticker = (
        trade.get(
            "ticker",
            "",
        )
        .strip()
        .upper()
    )

    trade["ticker"] = ticker

    if not TICKER_RE.match(
        ticker
    ):
        reject_trade(
            analysis,
            (
                "Ticker failed "
                "basic validation."
            ),
        )
        return

    causal_chain = trade.get(
        "causal_chain",
        []
    )

    if len(causal_chain) < 2:
        reject_trade(
            analysis,
            (
                "Causal chain was "
                "not sufficiently developed."
            ),
        )
        return

    evidence = trade.get(
        "evidence",
        []
    )

    if len(evidence) < 2:
        reject_trade(
            analysis,
            (
                "Fewer than two "
                "verification sources."
            ),
        )
        return

    if not any(
        item.get(
            "supports_company_exposure"
        )
        for item in evidence
    ):
        reject_trade(
            analysis,
            (
                "No evidence directly "
                "verified company exposure."
            ),
        )
        return

    consulted_urls = {
        normalize_url(
            source["url"]
        )
        for source
        in consulted_sources
    }

    verified_urls = {
        normalize_url(
            item.get(
                "source_url",
                "",
            )
        )
        for item in evidence
        if normalize_url(
            item.get(
                "source_url",
                "",
            )
        )
        in consulted_urls
    }

    # This is a useful anti-hallucination guard.
    #
    # Gilman doesn't merely have to print plausible URLs.
    # The URLs need to have actually appeared in the model's
    # web-search source list.
    if len(verified_urls) < 2:
        reject_trade(
            analysis,
            (
                "At least two stated "
                "evidence URLs were not found "
                "in the API web-search audit."
            ),
        )

# ============================================================

def deduplicate_trades(
    analyses: list[dict],
    stories_by_id: dict[str, dict],
) -> None:
    """
    If multiple headlines from the same persistent event produce
    the same ticker/direction thesis, keep the highest-confidence
    version rather than pretending they are independent trades.
    """

    groups = {}

    for analysis in analyses:
        if (
            analysis["decision"]
            != "TRADE"
        ):
            continue

        trade = analysis[
            "trade"
        ]

        story = stories_by_id[
            analysis["story_id"]
        ]

        event_id = (
            story[
                "event_context"
            ][
                "event_id"
            ]
        )

        key = (
            event_id,
            trade["ticker"],
            trade["direction"],
        )

        groups.setdefault(
            key,
            [],
        ).append(
            analysis
        )

    for group in groups.values():
        if len(group) <= 1:
            continue

        group.sort(
            key=lambda analysis: (
                analysis[
                    "trade"
                ][
                    "confidence"
                ]
            ),
            reverse=True,
        )

        winner = group[0]

        for duplicate in group[1:]:
            reject_trade(
                duplicate,
                (
                    "Duplicate thesis from "
                    f"the same event; retained "
                    f"{winner['story_id']}."
                ),
            )

            trades

        "no_trade_count": (

        "analyses": analyses,

    }

    trade_handoff = {
        "date": date_str,
        "generated_at": (
            result[
                "generated_at"
            ]

        "source": "gilman",
        "trade_count": len(

                    "trade"
                ],
            }
            for analysis
            in trades
        ],
    }

    dated_path = (

        "Wrote %s",

    )

    logger.info(
        "Wrote %s",
        GILMAN_TRADES_LATEST_PATH,
    )

def main():
    payload = load_input()

    stories = payload[
        "stories"
    ]

    stories_by_id = {
        story["story_id"]: story
        for story in stories
    }

    all_analyses = []
    web_audit = []

    for start in range(
        0,
        len(stories),
        BATCH_SIZE,
    ):
        batch = stories[
            start:start + BATCH_SIZE
        ]

        logger.info(
            "Gilman analyzing stories %d-%d of %d",
            start + 1,
            start + len(batch),
            len(stories),
        )

        (
            analyses,
            sources,
            response_id,
        ) = analyze_batch(
            batch
        )

        expected_ids = {
            story["story_id"]
            for story
            in batch
        }

        received_ids = {
            analysis["story_id"]
            for analysis
            in analyses
        }

        if (
            expected_ids
            != received_ids
        ):
            raise RuntimeError(
                "Gilman did not return "
                "exactly one analysis per story."
            )

        for analysis in analyses:
            validate_trade(
                analysis,
                sources,
            )

        all_analyses.extend(
            analyses
        )

        web_audit.append(
            {
                "batch_start_rank": (
                    batch[0]["rank"]
                ),
                "batch_end_rank": (
                    batch[-1]["rank"]
                ),
                "response_id": (
                    response_id
                ),
                "consulted_sources": (
                    sources
                ),
            }
        )

    deduplicate_trades(
        all_analyses,
        stories_by_id,
    )

    # Preserve original story ordering.
    order = {
        story["story_id"]: i
        for i, story
        in enumerate(stories)
    }

    all_analyses.sort(
        key=lambda analysis: (
            order[
                analysis[
                    "story_id"
                ]
            ]
        )
    )

    write_outputs(
        payload,
        all_analyses,
        web_audit,
    )


if __name__ == "__main__":
    main()

def clean_number(value):
    if value is None:
        return None

    try:
        value = float(value)

        if math.isnan(value) or math.isinf(value):
            return None

        return value

    except (TypeError, ValueError):
        return None


def pct(value):
    value = clean_number(value)

    if value is None:
        return None

    return round(value * 100, 3)


def return_pct(series: pd.Series, periods: int):
    series = series.dropna()

    if len(series) <= periods:
        return None

    old = float(series.iloc[-periods - 1])
    new = float(series.iloc[-1])

    if old == 0:
        return None

    return round(
        (new / old - 1) * 100,
        3,
    )


def sma_distance(close: pd.Series, window: int):
    close = close.dropna()

    if len(close) < window:
        return None

    current = float(close.iloc[-1])
    average = float(close.iloc[-window:].mean())

    if average == 0:
        return None

    return round(
        (current / average - 1) * 100,
        3,
    )


def realized_volatility(close: pd.Series, window: int = 20):
    close = close.dropna()

    returns = close.pct_change().dropna()

    if len(returns) < window:
        return None

    vol = (
        returns.iloc[-window:].std()
        * math.sqrt(252)
    )

    return round(
        float(vol) * 100,
        3,
    )


def get_daily_context(ticker: str) -> dict:
    data = yf.download(
        ticker,
        period="1y",
        interval="1d",
        auto_adjust=True,
        progress=False,
        threads=False,
        multi_level_index=False,
    )

    if data.empty:
        raise RuntimeError(
            f"No daily yfinance data for {ticker}"
        )

    close = data["Close"].dropna()
    volume = data["Volume"].dropna()

    current = float(close.iloc[-1])

    high_52w = float(
        data["High"].dropna().max()
    )

    low_52w = float(
        data["Low"].dropna().min()
    )

    if high_52w > low_52w:
        range_position = (
            (current - low_52w)
            / (high_52w - low_52w)
        ) * 100
    else:
        range_position = None

    volume_ratio = None

    if len(volume) >= 21:
        current_volume = float(volume.iloc[-1])
        normal_volume = float(
            volume.iloc[-21:-1].mean()
        )

        if normal_volume:
            volume_ratio = (
                current_volume
                / normal_volume
            )

    gap_pct = None

    if len(data) >= 2:
        previous_close = float(
            data["Close"].iloc[-2]
        )

        today_open = float(
            data["Open"].iloc[-1]
        )

        if previous_close:
            gap_pct = (
                today_open
                / previous_close
                - 1
            ) * 100

    return {
        "last_price": round(current, 4),

        "returns_pct": {
            "1d": return_pct(close, 1),
            "5d": return_pct(close, 5),
            "20d": return_pct(close, 20),
            "60d": return_pct(close, 60),
            "120d": return_pct(close, 120),
            "252d": return_pct(close, 252),
        },

        "gap_today_pct": (
            round(gap_pct, 3)
            if gap_pct is not None
            else None
        ),

        "volume_vs_20d_average": (
            round(volume_ratio, 3)
            if volume_ratio is not None
            else None
        ),

        "realized_volatility_20d_pct": (
            realized_volatility(
                close,
                20,
            )
        ),

        "moving_average_distance_pct": {
            "20d": sma_distance(
                close,
                20,
            ),
            "50d": sma_distance(
                close,
                50,
            ),
            "200d": sma_distance(
                close,
                200,
            ),
        },

        "52_week": {
            "high": round(
                high_52w,
                4,
            ),
            "low": round(
                low_52w,
                4,
            ),
            "below_high_pct": round(
                (
                    current
                    / high_52w
                    - 1
                )
                * 100,
                3,
            ),
            "range_position_pct": (
                round(
                    range_position,
                    3,
                )
                if range_position is not None
                else None
            ),
        },
    }


def get_intraday_context(ticker: str) -> dict:

        data = yf.download(

            period="5d",
            interval="5m",
            auto_adjust=True,
            prepost=False,
            progress=False,
            threads=False,
            multi_level_index=False,
        )

        if data.empty:
            return {}

        # Use the most recent trading session in the dataset.
        last_date = data.index[-1].date()

        today = data[
            data.index.date == last_date
        ]

        if today.empty:
            return {}

        session_open = float(
            today["Open"].iloc[0]
        )

        latest = float(
            today["Close"].iloc[-1]
        )

        high = float(
            today["High"].max()
        )

        low = float(
            today["Low"].min()
        )

        move = None

        if session_open:
            move = (
                latest
                / session_open
                - 1
            ) * 100

        return {
            "session_date": (
                last_date.isoformat()
            ),
            "session_open": round(
                session_open,

            "latest_price": round(
                latest,

            "session_high": round(
                high,
                4,
            ),
            "session_low": round(
                low,
                4,
            ),
            "from_open_pct": (
                round(move, 3)
                if move is not None
                else None
            ),
        }

    except Exception:
        logger.exception(
            "Intraday data failed for %s",
            ticker,

        return {}

    obj = yf.Ticker(ticker)

    try:
        info = obj.get_info()
    except Exception:
        logger.exception(
            "Ticker.info failed for %s",
            ticker,
        )
        info = {}

    result = {}

    for field in INFO_FIELDS:
        value = info.get(field)

        if field in {
            "sector",
            "industry",
        }:
            result[field] = value
            continue

        if field in PERCENT_INFO_FIELDS:
            result[field] = pct(value)
        else:
            result[field] = clean_number(
                value
            )

    return result


def get_spy_returns() -> dict:
    data = yf.download(
        "SPY",
        period="1y",
        interval="1d",
        auto_adjust=True,
        progress=False,
        threads=False,
        multi_level_index=False,
    )

    if data.empty:
        return {}

    close = data["Close"].dropna()

    return {
        "1d": return_pct(close, 1),
        "5d": return_pct(close, 5),
        "20d": return_pct(close, 20),
        "60d": return_pct(close, 60),
    }


def relative_return(
    stock_return,
    benchmark_return,
):
    if (
        stock_return is None
        or benchmark_return is None
    ):
        return None

    return round(
        stock_return - benchmark_return,
        3,
    )


def build_context(
    trade: dict,
    spy_returns: dict,
) -> dict:

    ticker = trade["ticker"].upper()

    logger.info(
        "Collecting market context for %s",
        ticker,
    )

    daily = get_daily_context(
        ticker
    )

    intraday = get_intraday_context(
        ticker
    )

    fundamentals = get_fundamentals(
        ticker
    )

    stock_returns = daily[
        "returns_pct"
    ]

    relative = {}

    for horizon in [
        "1d",
        "5d",
        "20d",
        "60d",
    ]:
        relative[horizon] = (
            relative_return(
                stock_returns.get(
                    horizon
                ),
                spy_returns.get(
                    horizon
                ),
            )
        )

    return {
        "gilman_trade": trade,

        "market_data": {
            "daily": daily,
            "intraday": intraday,
            "relative_to_spy_pct": (
                relative
            ),
        },

        "fundamentals": (
            fundamentals
        ),
    }


def main():
    with INPUT_PATH.open(
        "r",
        encoding="utf-8",
    ) as f:
        gilman = json.load(f)

    trades = gilman.get(
        "trades",
        []
    )

    logger.info(
        "Gilman supplied %d trades",
        len(trades),
    )

    spy_returns = get_spy_returns()

    candidates = []

    for trade in trades:
        try:
            candidates.append(
                build_context(
                    trade,
                    spy_returns,
                )
            )

        except Exception as exc:
            logger.exception(
                "Market context failed for %s",
                trade.get(
                    "ticker"
                ),
            )

            candidates.append(
                {
                    "gilman_trade": trade,
                    "market_data_error": (
                        str(exc)
                    ),
                }
            )

    output = {
        "date": gilman["date"],
        "generated_at": utc_now(),
        "candidate_count": len(
            candidates
        ),
        "benchmark": {
            "ticker": "SPY",
            "returns_pct": (
                spy_returns
            ),
        },
        "candidates": candidates,
    }

    OUTPUT_PATH.write_text(
        json.dumps(
            output,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

    logger.info(
        "Wrote %s",
        OUTPUT_PATH,
    )

    main()

def clean_whitespace(value: str) -> str:
    return re.sub(r"\s+", " ", value).strip()


def strip_html(value: str | None) -> str:
    """
    Turn an RSS HTML description into plain text.
    """

    if not value:
        return ""

    value = re.sub(
        r"<script\b[^>]*>.*?</script>",
        " ",
        value,
        flags=re.IGNORECASE | re.DOTALL,
    )

    value = re.sub(
        r"<style\b[^>]*>.*?</style>",
        " ",
        value,
        flags=re.IGNORECASE | re.DOTALL,
    )

    value = re.sub(
        r"<[^>]+>",
        " ",
        value,
    )

    value = html.unescape(value)

    return clean_whitespace(value)


def clean_url(url: str) -> str:
    """
    Remove fragments and common tracking query strings.

    Google News links are intentionally retained as Google News URLs;
    resolving every redirect is not necessary for clustering.
    """

    if not url:
        return ""

    parts = urlsplit(url)

    hostname = parts.netloc.lower()

    # Google News article URLs often require their query information,
    # so don't strip it aggressively.
    if "news.google.com" in hostname:
        return urlunsplit(
            (
                parts.scheme,
                hostname,
                parts.path.rstrip("/"),
                parts.query,
                "",
            )
        )

    return urlunsplit(
        (
            parts.scheme,
            hostname,
            parts.path.rstrip("/"),
            "",
            "",
        )
    )


def normalized_title(title: str) -> str:
    """
    Normalize a title for within-source duplicate detection.
    """

    title = html.unescape(title)
    title = title.casefold()

    title = re.sub(
        r"[^a-z0-9]+",
        " ",
        title,
    )

    return clean_whitespace(title)


def canonicalize_source(source: str) -> str:
    """
    Avoid counting small naming variations as separate outlets.
    """

    source = clean_whitespace(source)

    aliases = {
        "bbc": "BBC News",
        "bbc news": "BBC News",
        "the guardian": "The Guardian",
        "guardian": "The Guardian",
        "new york times": "The New York Times",
        "the new york times": "The New York Times",
        "nyt": "The New York Times",
        "npr": "NPR",
        "cnn": "CNN",
        "ap": "AP News",
        "ap news": "AP News",
        "associated press": "AP News",
        "al jazeera": "Al Jazeera",
        "al jazeera english": "Al Jazeera",
    }

    return aliases.get(
        source.casefold(),
        source,
    )


def remove_source_suffix(
    title: str,
    source: str,
) -> str:
    """
    Google News commonly gives titles like:

        Something happened - Reuters

    Since Google also exposes Reuters as the source field, remove that
    suffix before embedding.
    """

    title = clean_whitespace(title)

    if not source:
        return title

    for separator in (" - ", " – ", " — "):
        suffix = f"{separator}{source}"

        if title.casefold().endswith(suffix.casefold()):
            title = title[:-len(suffix)].strip()
            break

    return title

def struct_time_to_datetime(
    value,
) -> datetime | None:

    if value is None:
        return None

    try:
        timestamp = calendar.timegm(value)
        return datetime.fromtimestamp(
            timestamp,
            tz=timezone.utc,
        )
    except (TypeError, ValueError, OverflowError):
        return None


def entry_datetime(entry) -> datetime | None:
    """
    feedparser normalizes RSS/Atom timestamps into *_parsed values.
    """

    published = struct_time_to_datetime(
        entry.get("published_parsed")
    )

    if published:
        return published

    updated = struct_time_to_datetime(
        entry.get("updated_parsed")
    )

    if updated:
        return updated

    created = struct_time_to_datetime(
        entry.get("created_parsed")
    )

    return created

def get_entry_source(
    entry,
    feed_config: dict,
) -> str:

    if not feed_config["aggregator"]:
        return canonicalize_source(
            feed_config["source"]
        )

    # Google News RSS normally provides:
    #
    # <source url="...">Reuters</source>
    #
    source = entry.get("source")

    if isinstance(source, dict):
        source_title = source.get("title")

        if source_title:
            return canonicalize_source(
                strip_html(source_title)
            )

    return canonicalize_source(
        feed_config["source"]
    )


def fetch_feed(
    feed_config: dict,
) -> list[dict]:

    logger.info(
        "Fetching RSS: %s",
        feed_config["name"],
    )

    response = session.get(
        feed_config["url"],
        timeout=REQUEST_TIMEOUT,
    )

    response.raise_for_status()

    parsed = feedparser.parse(
        response.content
    )

    if parsed.bozo:
        logger.warning(
            "RSS parser warning for %s: %s",
            feed_config["name"],
            parsed.bozo_exception,
        )

    articles = []

    entries = parsed.entries[
        :MAX_ITEMS_PER_FEED
    ]

    for entry in entries:
        raw_title = strip_html(
            entry.get("title")
        )

        if not raw_title:
            continue

        source = get_entry_source(
            entry,
            feed_config,
        )

        title = raw_title

        if feed_config["aggregator"]:
            title = remove_source_suffix(
                title,
                source,
            )

        description = strip_html(
            entry.get("summary")
            or entry.get("description")
            or ""
        )

        url = (
            entry.get("link")
            or entry.get("id")
            or ""
        )

        published_dt = entry_datetime(
            entry
        )

        articles.append(
            {
                "title": title,
                "description": description,
                "url": url,
                "clean_url": clean_url(url),
                "source": source,
                "published_at": (
                    published_dt.isoformat()
                    if published_dt
                    else None
                ),
                "_published_dt": published_dt,
                "feed_name": feed_config["name"],
                "feed_url": feed_config["url"],
                "category": feed_config["category"],
                "region": feed_config["region"],
                "aggregator": feed_config["aggregator"],
            }
        )

    logger.info(
        "Fetched %d entries from %s",
        len(articles),
        feed_config["name"],
    )

    return articles


def collect_articles() -> list[dict]:
    """
    Fetch all RSS feeds.

    A failed feed does NOT kill the daily job. It is logged and the
    remaining feeds continue.
    """

    raw_articles = []

    successful_feeds = 0
    failed_feeds = 0

    for feed_config in RSS_FEEDS:
        try:
            articles = fetch_feed(
                feed_config
            )

            raw_articles.extend(
                articles
            )

            successful_feeds += 1

        except Exception:
            failed_feeds += 1

            logger.exception(
                "Failed RSS feed: %s",
                feed_config["name"],
            )

    logger.info(
        "RSS collection complete: "
        "%d successful feeds, "
        "%d failed feeds, "
        "%d raw entries",
        successful_feeds,
        failed_feeds,
        len(raw_articles),
    )

    if successful_feeds == 0:
        raise RuntimeError(
            "Every RSS feed failed."
        )

    cutoff = (
        datetime.now(timezone.utc)
        - timedelta(hours=LOOKBACK_HOURS)
    )

    # --------------------------------------------------------
    # Deduplication
    #
    # IMPORTANT:
    #
    # We don't globally dedupe identical titles because identical
    # coverage from two DIFFERENT outlets is exactly the signal we
    # want for measuring story importance.
    #
    # Instead we remove duplicates from the SAME publisher.
    # --------------------------------------------------------

    by_source_and_title: dict[
        tuple[str, str],
        dict,
    ] = {}

    skipped_old = 0
    skipped_undated = 0

    for article in raw_articles:
        title = article["title"].strip()

        if not title:
            continue

        published_dt = article["_published_dt"]

        if published_dt is None:
            if not ALLOW_UNDATED_ARTICLES:
                skipped_undated += 1
                continue

        elif published_dt < cutoff:
            skipped_old += 1
            continue

        source_key = article[
            "source"
        ].casefold()

        title_key = normalized_title(
            title
        )

        if not title_key:
            continue

        dedupe_key = (
            source_key,
            title_key,
        )

        existing = by_source_and_title.get(
            dedupe_key
        )

        if existing is None:
            article["categories"] = {
                article["category"]
            }

            article["regions"] = {
                article["region"]
            }

            article["feed_names"] = {
                article["feed_name"]
            }

            by_source_and_title[
                dedupe_key
            ] = article

            continue

        # Same publisher/title found in several RSS feeds.
        # Merge metadata rather than treating it as separate coverage.
        existing["categories"].add(
            article["category"]
        )

        existing["regions"].add(
            article["region"]
        )

        existing["feed_names"].add(
            article["feed_name"]
        )

        # Prefer whichever record has a richer description.
        if (
            len(article["description"])
            > len(existing["description"])
        ):
            existing["description"] = (
                article["description"]
            )

        # Prefer a direct publisher link over a Google News link.
        if (
            existing["aggregator"]
            and not article["aggregator"]
        ):
            existing["url"] = article["url"]
            existing["clean_url"] = (
                article["clean_url"]
            )
            existing["aggregator"] = False

        # Keep newest timestamp.
        existing_dt = existing[
            "_published_dt"
        ]

        article_dt = article[
            "_published_dt"
        ]

        if (
            article_dt is not None
            and (
                existing_dt is None
                or article_dt > existing_dt
            )
        ):
            existing["_published_dt"] = (
                article_dt
            )

            existing["published_at"] = (
                article_dt.isoformat()
            )

    articles = list(
        by_source_and_title.values()
    )

    for article in articles:
        article["categories"] = sorted(
            article["categories"]
        )

        article["regions"] = sorted(
            article["regions"]
        )

        article["feed_names"] = sorted(
            article["feed_names"]
        )

        # Internal datetime is no longer needed.
        article.pop(
            "_published_dt",
            None,
        )

        # Remove redundant single-value fields.
        article.pop(
            "category",
            None,
        )

        article.pop(
            "region",
            None,
        )

        article.pop(
            "feed_name",
            None,
        )

    logger.info(
        "%d unique recent publisher articles remain "
        "after filtering/deduplication",
        len(articles),
    )

    logger.info(
        "Skipped %d old entries and %d undated entries",
        skipped_old,
        skipped_undated,
    )

    return articles

def text_for_embedding(
    article: dict,
) -> str:

    title = article["title"]

    description = (
        article.get("description")
        or ""
    )

    # Enough context to distinguish similar headlines while
    # preventing giant RSS descriptions from dominating.
    description = description[:500]

    return (
        f"Headline: {title}\n"
        f"Description: {description}"
    )


def create_embeddings(
    articles: list[dict],
    batch_size: int = 200,
) -> np.ndarray:

    texts = [
        text_for_embedding(article)
        for article in articles
    ]

    vectors = []

    for start in range(
        0,
        len(texts),
        batch_size,
    ):
        batch = texts[
            start:start + batch_size
        ]

        logger.info(
            "Embedding articles %d-%d of %d",
            start + 1,
            start + len(batch),
            len(texts),
        )

        response = (
            openai_client.embeddings.create(
                model=EMBEDDING_MODEL,
                input=batch,
            )
        )

        ordered = sorted(
            response.data,
            key=lambda item: item.index,
        )

        vectors.extend(
            item.embedding
            for item in ordered
        )

    return np.asarray(
        vectors,
        dtype=np.float32,
    )

def parse_iso_datetime(
    value: str | None,
) -> datetime | None:

    if not value:
        return None

    try:
        return datetime.fromisoformat(
            value.replace(
                "Z",
                "+00:00",
            )
        )

    except ValueError:
        return None


def cluster_articles(
    articles: list[dict],
    vectors: np.ndarray,
) -> list[dict]:

    if len(articles) == 1:
        labels = np.array([0])

    else:
        logger.info(
            "Clustering %d articles",
            len(articles),
        )

        clusterer = AgglomerativeClustering(
            n_clusters=None,
            distance_threshold=(
                CLUSTER_DISTANCE_THRESHOLD
            ),
            metric="cosine",
            linkage="average",
        )

        labels = clusterer.fit_predict(
            vectors
        )

    grouped = defaultdict(list)

    for article, label in zip(
        articles,
        labels,
    ):
        grouped[
            int(label)
        ].append(article)

    clusters = []

    now = datetime.now(
        timezone.utc
    )

    for (
        cluster_id,
        cluster_articles_,
    ) in grouped.items():

        sources = sorted(
            {
                article["source"]
                for article
                in cluster_articles_
            }
        )

        regions = sorted(
            {
                region
                for article
                in cluster_articles_
                for region
                in article["regions"]
            }
        )

        categories = sorted(
            {
                category
                for article
                in cluster_articles_
                for category
                in article["categories"]
            }
        )

        publication_times = [
            parse_iso_datetime(
                article[
                    "published_at"
                ]
            )
            for article
            in cluster_articles_
        ]

        publication_times = [
            dt
            for dt
            in publication_times
            if dt is not None
        ]

        newest = (
            max(publication_times)
            if publication_times
            else now
        )

        age_hours = max(
            0,
            (
                now - newest
            ).total_seconds()
            / 3600,
        )

        # ----------------------------------------------------
        # IMPORTANCE SCORE
        #
        # The biggest signal is independent publisher coverage.
        #
        # A story appearing at Reuters/AP/BBC/CNN/etc. should beat
        # something appearing twenty times from one publisher.
        #
        # Freshness matters, but not so much that a trivial story
        # from 10 minutes ago automatically beats major news from
        # 8 hours ago.
        # ----------------------------------------------------

        source_score = (
            7.0
            * math.log1p(
                len(sources)
            )
        )

        article_score = (
            1.0
            * math.log1p(
                len(
                    cluster_articles_
                )
            )
        )

        category_score = (
            1.0
            * math.log1p(
                len(categories)
            )
        )

        region_score = (
            0.75
            * math.log1p(
                len(regions)
            )
        )

        freshness_score = (
            3.0
            * math.exp(
                -age_hours / 18.0
            )
        )

        score = (
            source_score
            + article_score
            + category_score
            + region_score
            + freshness_score
        )

        clusters.append(
            {
                "cluster_id": (
                    cluster_id
                ),
                "score": round(
                    score,
                    4,
                ),
                "source_count": len(
                    sources
                ),
                "article_count": len(
                    cluster_articles_
                ),
                "sources": sources,
                "regions": regions,
                "categories": (
                    categories
                ),
                "newest_published_at": (
                    newest.isoformat()
                ),
                "articles": (
                    cluster_articles_
                ),
            }
        )

    clusters.sort(
        key=lambda cluster: (
            cluster["score"]
        ),
        reverse=True,
    )

    logger.info(
        "%d distinct story clusters found",
        len(clusters),
    )

    return clusters

SUMMARY_SCHEMA = {
    "type": "object",
    "properties": {
        "stories": {
            "type": "array",
            "items": {
                "type": "object",
                "properties": {
                    "cluster_id": {
                        "type": "integer",
                    },
                    "title": {
                        "type": "string",
                    },
                    "description": {
                        "type": "string",
                    },
                },
                "required": [
                    "cluster_id",
                    "title",
                    "description",
                ],
                "additionalProperties": (
                    False
                ),
            },
        }
    },
    "required": [
        "stories"
    ],
    "additionalProperties": False,
}


def cluster_for_prompt(
    cluster: dict,
) -> dict:

    # Prefer articles with descriptions and recent timestamps.
    articles = sorted(
        cluster["articles"],
        key=lambda x: (
            bool(
                x.get(
                    "description"
                )
            ),
            x.get(
                "published_at"
            )
            or "",
        ),
        reverse=True,
    )

    evidence = []

    # Eight different articles is generally more than enough
    # evidence to summarize one event.
    for article in articles[:8]:
        evidence.append(
            {
                "source": (
                    article["source"]
                ),
                "headline": (
                    article["title"]
                ),
                "description": (
                    article[
                        "description"
                    ][:500]
                ),
                "published_at": (
                    article[
                        "published_at"
                    ]
                ),
            }
        )

    return {
        "cluster_id": (
            cluster[
                "cluster_id"
            ]
        ),
        "source_count": (
            cluster[
                "source_count"
            ]
        ),
        "sources": (
            cluster["sources"]
        ),
        "articles": evidence,
    }


def summarize_batch(
    clusters: list[dict],
) -> list[dict]:

    payload = {
        "stories": [
            cluster_for_prompt(
                cluster
            )
            for cluster
            in clusters
        ]
    }

    instructions = """
You are an editor producing a daily list of major news stories.

Each input cluster represents RSS headlines that an automated
clustering system believes concern the same underlying news event.

For every cluster:

1. Produce one neutral canonical headline.
2. Produce one concise one-sentence description explaining what
   happened and why the development is notable.
3. Base the output ONLY on facts supported by the supplied headlines
   and descriptions.
4. Do not invent facts, numbers, quotations, names, motives, causes,
   consequences, dates, or background.
5. When evidence is incomplete, describe only what is clearly supported.
6. When sources disagree, use cautious wording and do not resolve the
   disagreement yourself.
7. Do not use sensational or clickbait wording.
8. Do not mention the RSS feed, cluster, ranking system, or source count
   in the finished headline or description.
9. Headline should preferably contain 8-18 words.
10. Description should preferably contain 20-45 words.
11. Preserve cluster_id exactly.
12. Return every supplied cluster exactly once.
"""

    response = (
        openai_client.responses.create(
            model=SUMMARY_MODEL,
            instructions=instructions,
            input=json.dumps(
                payload,
                ensure_ascii=False,
            ),
            reasoning={
                "effort": "low",
            },
            text={
                "format": {
                    "type": (
                        "json_schema"
                    ),
                    "name": (
                        "daily_news_summaries"
                    ),
                    "strict": True,
                    "schema": (
                        SUMMARY_SCHEMA
                    ),
                }
            },
            max_output_tokens=8000,
            store=False,
        )
    )

    if not response.output_text:
        raise RuntimeError(
            "OpenAI returned no output text."
        )

    result = json.loads(
        response.output_text
    )

    return result[
        "stories"
    ]


def summarize_clusters(
    clusters: list[dict],
) -> dict[int, dict]:

    results = {}

    for start in range(
        0,
        len(clusters),
        SUMMARY_BATCH_SIZE,
    ):
        batch = clusters[
            start:
            start + SUMMARY_BATCH_SIZE
        ]

        logger.info(
            "Summarizing stories %d-%d of %d",
            start + 1,
            start + len(batch),
            len(clusters),
        )

        summaries = (
            summarize_batch(
                batch
            )
        )

        for summary in summaries:
            results[
                summary[
                    "cluster_id"
                ]
            ] = summary

    return results

def write_outputs(
    top_clusters: list[dict],
    summaries: dict[int, dict],
) -> None:

    generated_at = datetime.now(
        timezone.utc
    )

    stories = []
    debug_stories = []

    for rank, cluster in enumerate(
        top_clusters,
        start=1,
    ):
        summary = summaries.get(
            cluster[
                "cluster_id"
            ]
        )

        if summary is None:
            logger.warning(
                "Missing summary for cluster %s",
                cluster[
                    "cluster_id"
                ],
            )

            continue

        public_story = {
            "rank": rank,
            "title": (
                summary["title"]
            ),
            "description": (
                summary[
                    "description"
                ]
            ),
        }

        stories.append(
            public_story
        )

        debug_stories.append(
            {
                **public_story,
                "score": (
                    cluster["score"]
                ),
                "source_count": (
                    cluster[
                        "source_count"
                    ]
                ),
                "article_count": (
                    cluster[
                        "article_count"
                    ]
                ),
                "sources": (
                    cluster[
                        "sources"
                    ]
                ),
                "regions": (
                    cluster[
                        "regions"
                    ]
                ),
                "categories": (
                    cluster[
                        "categories"
                    ]
                ),
                "articles": [
                    {
                        "title": (
                            article[
                                "title"
                            ]
                        ),
                        "source": (
                            article[
                                "source"
                            ]
                        ),
                        "url": (
                            article[
                                "url"
                            ]
                        ),
                        "published_at": (
                            article[
                                "published_at"
                            ]
                        ),
                        "feed_names": (
                            article[
                                "feed_names"
                            ]
                        ),
                    }
                    for article
                    in cluster[
                        "articles"
                    ]
                ],
            }
        )

    public_output = {
        "date": (
            generated_at
            .date()
            .isoformat()
        ),
        "generated_at": (
            generated_at.isoformat()
        ),
        "window_hours": (
            LOOKBACK_HOURS
        ),
        "story_count": len(
            stories
        ),
        "stories": stories,
    }

    debug_output = {
        **public_output,
        "stories": debug_stories,
    }

    date_str = (
        generated_at
        .date()
        .isoformat()
    )

    dated_path = (
        DATA_DIR
        / f"{date_str}.json"
    )

    latest_path = (
        DATA_DIR
        / "latest.json"
    )

    debug_path = (
        DATA_DIR
        / f"{date_str}.debug.json"
    )

    dated_path.write_text(
        json.dumps(
            public_output,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

    debug_path.write_text(
        json.dumps(
            debug_output,
            indent=2,
            ensure_ascii=False,
        )
        + "\n",
        encoding="utf-8",
    )

    shutil.copyfile(
        dated_path,
        latest_path,
    )

    logger.info(
        "Wrote %s",
        dated_path,
    )

    logger.info(
        "Wrote %s",
        latest_path,
    )

    logger.info(
        "Wrote %s",
        debug_path,
    )

def print_top_clusters(
    clusters: list[dict],
    limit: int = 20,
) -> None:

    logger.info(
        "Top %d clusters before GPT summarization:",
        min(
            limit,
            len(clusters),
        ),
    )

    for rank, cluster in enumerate(
        clusters[:limit],
        start=1,
    ):
        representative = max(
            cluster["articles"],
            key=lambda article: (
                len(
                    article[
                        "description"
                    ]
                )
            ),
        )

        logger.info(
            "#%d | score=%.2f | sources=%d | articles=%d | %s",
            rank,
            cluster["score"],
            cluster[
                "source_count"
            ],
            cluster[
                "article_count"
            ],
            representative[
                "title"
            ],
        )

# MAIN
# ============================================================

def main():
    logger.info(
        "Starting daily RSS news update"
    )

    logger.info(
        "Looking back %d hours",
        LOOKBACK_HOURS,
    )

    logger.info(
        "Configured RSS feeds: %d",
        len(RSS_FEEDS),
    )

    # --------------------------------------------------------
    # 1. COLLECT RSS
    # --------------------------------------------------------

    articles = collect_articles()

    if (
        len(articles)
        < TARGET_STORIES
    ):
        raise RuntimeError(
            f"Only {len(articles)} recent "
            "candidate articles found; "
            f"need at least {TARGET_STORIES}."
        )

    # --------------------------------------------------------
    # 2. EMBED
    # --------------------------------------------------------

    vectors = create_embeddings(
        articles
    )

    # --------------------------------------------------------
    # 3. CLUSTER
    # --------------------------------------------------------

    clusters = cluster_articles(
        articles,
        vectors,
    )

    print_top_clusters(
        clusters
    )

    if (
        len(clusters)
        < TARGET_STORIES
    ):
        raise RuntimeError(
            f"Only {len(clusters)} distinct "
            "story clusters found. "
            "Try lowering "
            "CLUSTER_DISTANCE_THRESHOLD "
            "or increasing LOOKBACK_HOURS."
        )

    # --------------------------------------------------------
    # 4. SELECT TOP 100
    # --------------------------------------------------------

    top_clusters = clusters[
        :TARGET_STORIES
    ]

    # --------------------------------------------------------
    # 5. GPT SUMMARIES
    # --------------------------------------------------------

    summaries = summarize_clusters(
        top_clusters
    )

    # --------------------------------------------------------
    # 6. WRITE JSON
    # --------------------------------------------------------

    write_outputs(
        top_clusters,
        summaries,
    )

    logger.info(
        "Daily RSS news update complete"
    )


if __name__ == "__main__":
    main()
